$001B - [SOUND_SEQ_POINTER7 (High)]

$002C - [GENERAL_MEMORY_POINTER (Low)]
$002D - [GENERAL_MEMORY_POINTER (High)]
$002E - [MEMMOVE_DESTINATION_ADDRESS (Low)] - The MemoryMove command stores the address of where it is moving memory to here.
$002F - [MEMMOVE_DESTINATION_ADDRESS (High)]

$005D = [V0SRCN_REGISTER_VALUE]
$005E = [V1SRCN_REGISTER_VALUE]

$0062 = [V5SRCN_REGISTER_VALUE]
$0063 = [V6SRCN_REGISTER_VALUE]
$0064 = [V7SRCN_REGISTER_VALUE]

$00A7 - [EVOL_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [EVOLL] & [EVOLR] DSP Registers.

$00AF - [KON_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [KON] DSP Register.
$00B0 - [KOF_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [KOF] DSP Register.

$00B4 - [CURRENT_CPUI0_READ] - The last value read out of the [CPUI0] Register.
$00B5 - [CURRENT_CPUI1_READ] - The last value read out of the [CPUI1] Register.
$00B6 - [CURRENT_CPUI2_READ] - The last value read out of the [CPUI2] Register.
$00B7 - [CURRENT_CPUI3_READ] - The last value read out of the [CPUI3] Register.
$00B8 - [EON_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [EON] DSP Register.
$00B9 - [NON_REGISTER_VALUE] - Set by (($AE ^ 0xFF) & $BE) | $BF
$00BA - [PMON_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [PMON] DSP Register.
$00BB - [FLG_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [FLG] DSP Register.

$00C7 - [FILTER_COEF_INDEX]
$00C8 - [EFB_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [EFB] DSP Register.

$00DD - Seems to be a timer. Calls a routine every 7 ticks.

$00DF - [EDL_REGISTER_VALUE] - The value the AkaoSpcEngine will place in the [EDL] DSP Register.
$00E0 - [PREVIOUS_EDL_VALUE]
$00E1 - [ECHO_WRITE_BUFFER_ADDRESS (Low)] - The address of the active echo buffer.
$00E2 - [ECHO_WRITE_BUFFER_ADDRESS (High)]
$00E3 - Unused?
$00E4 - Unused?
$00E5 - Unused?
$00E6 - Unused?
$00E7 - Unused?
$00E8 - Unused?
$00E9 - Unused?
$00EA - Unused?
$00EB - Unused?
$00EC - Unused?
$00ED - Unused?
$00EE - Unused?
$00EF - Unused?
$00F0 - [TEST]          - Testing Functions Register
$00F1 - [CONTROL]       - I/O and Timer Control Register
$00F2 - [DSPADDR]       - DSP Communication Address Register
$00F3 - [DSPDATA]       - DSP Connunication Data Register
$00F4 - [CPUI0]/[CPUO0] - CPU Input/Output 0 Register
$00F5 - [CPUI1]/[CPUO1] - CPU Input/Output 1 Register
$00F6 - [CPUI2]/[CPUO2] - CPU Input/Output 2 Register
$00F7 - [CPUI3]/[CPUO3] - CPU Input/Output 3 Register
$00F8 - [AUXIO4]        - External I/O Port P4 Register (Disconnected/Unused)
$00F9 - [AUXIO5]        - External I/O Port P5 Register (Disconnected/Unused)
$00FA - [T0TARGET]      - Timer 0 Scaling Target Register
$00FB - [T1TARGET]      - Timer 1 Scaling Target Register
$00FC - [T2TARGET]      - Timer 2 Scaling Target Register
$00FD - [T0OUT]         - Timer 0 Output Register
$00FE - [T1OUT]         - Timer 1 Output Register
$00FF - [T2OUT]         - Timer 2 Output Register
$0100-$01FF             - Program Stack
$0200-$1417             - Program Code
$1418-$1597             - Pointer Tables & DSP Data
$1598-$17FF - Unused?
$1800-$1817 - SFX related array
$1818-$187F - ????
$1880-$1897 - SFX related array
$1898-$18FF - ????
$1900-$192B - Sample Source Directory

$2C00-$47FF - SFX Data?
$4800-$4F00 - BRR Sample Data

$FE80-$FF8F-[VxADSR_REGISTER_VALUES] - Two bytes per voice. Byte 1 is VxADSR1 and Byte 2 is VxADSR2.

$FFC0-FFFF - IPL ROM

[InitializeAkaoSpcEngine]
00/0200: 20              CLRP                ;Clear Direct Page Flag.
00/0201: C0              DI                  ;Disable Interrupts. (But why tho, there aren't any to disable)
00/0202: CDFF            MOV X, #$FF         ;Move 0xFF into Register X.
00/0204: BD              MOV SP, X           ;Move Register X to the Stack Pointer.
00/0205: E800            MOV A, #$00         ;Move 0x00 into the Accumulator.
00/0207: 5D              MOV X, A            ;Move 0x00 into Register X.

[ClearPageZeroMemory]
00/0208: AF              MOV (X)+, A         ;Move 0x00 into [$0000, X] and increment Register X.
00/0209: C8F0            CMP X, #$F0         ;Compare Register X against 0xF0.
00/020B: D0FB            BNE $0208           ;Branch to ClearPageZeroMemory and keep clearing $0000-$00F0.
00/020D: E800            MOV A, #$00         ;Move 0x00 into the Accumulator. (Preparing to zero out a bunch of DSP Registers)
00/020F: 8D0C            MOV Y, #$0C         ;Load [MVOLL] into Register Y.
00/0211: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0214: 8D1C            MOV Y, #$1C         ;Load [MVOLR] into Register Y.
00/0216: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0219: 8D2C            MOV Y, #$2C         ;Load [EVOLL] into Register Y.
00/021B: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/021E: 8D3C            MOV Y, #$3C         ;Load [EVOLR] into Register Y.
00/0220: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0223: 8D2D            MOV Y, #$2D         ;Load [PMON] into Register Y.
00/0225: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0228: 8D3D            MOV Y, #$3D         ;Load [NON] into Register Y.
00/022A: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/022D: 8D4D            MOV Y, #$4D         ;Load [EON] into Register Y.
00/022F: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0232: 8D5D            MOV Y, #$5D         ;Load [DIR] into Register Y.
00/0234: E819            MOV A, #$19         ;Load 0x19 into the Accumulator. (Set the start address for the samples pointer table at 00/1900)
00/0236: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0239: 8D07            MOV Y, #$07         ;Load [GAIN0] into Register Y.
00/023B: CDA0            MOV X, #$A0         ;Load 0xA0 into Register X.  (Decrease Exponential: INFINITE)

[InitializeVoiceGain]
00/023D: CBF2            MOV $F2, Y          ;Move [VxGAIN] into [DSPADDR].
00/023F: D8F3            MOV $F3, X          ;Move 0xA0 to [DSPDATA].
00/0241: DD              MOV A, Y            ;Move [VxGAIN] from Register Y to the Accumulator.
00/0242: 60              CLRC                ;Clear Carry Flag.
00/0243: 8810            ADC A, #$10         ;Add 0x10 to the value of [VxGAIN] to move to the next voice.
00/0245: FD              MOV Y, A            ;Move the [VxGAIN] register Id back to Register Y.
00/0246: 10F5            BPL $023D           ;Branch to InitializeVoiceGain if the Register Value is <0x87.
00/0248: 8FF0F1          MOV $F1, #$F0       ;Move 0xF0 into [CONTROL]. (Read IPL ROM and set Input Ports to zero)
00/024B: 8F24FA          MOV $FA, #$24       ;Move 0x24 to [T0TARGET]. (Timer1 increments every 0x24 ticks)
00/024E: 8F01F1          MOV $F1, #$01       ;Move 0x01 into [CONTROL]. (Enable Timer1).
00/0251: 8F02DF          MOV $DF, #$02       ;Move 0x02 into [EDL_REGISTER_VALUE].
00/0254: 3FDE12          CALL $12DE          ;Call InitializeDSPEchoRegisters.
00/0257: 3F2F13          CALL $132F          ;Call InitializeEchoBuffers.
00/025A: E87F            MOV A, #$7F         ;Load 0x7F into the Accumulator. (Max sound volume)
00/025C: 8D0C            MOV Y, #$0C         ;Load [MVOLL] into Register Y.
00/025E: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0261: 8D1C            MOV Y, #$1C         ;Load [MVOLR] into Register Y.
00/0263: 3F1B06          CALL $061B          ;Call SetDSPRegisterValue (Y = Register, A = Data).
00/0266: 8FFFB0          MOV $B0, #$FF       ;Move 0xFF into [KOF_REGISTER_VALUE]. (Fade out all voices)
00/0269: 8F07DD          MOV $DD, #$07
00/026C: E290            SET7 $90
00/026E: E29C            SET7 $9C
00/0270: 8D08            MOV Y, #$08
00/0272: E8FF            MOV A, #$FF
00/0274: D67FFF          MOV $FF7F+Y, A
00/0277: FEFB            DBNZ Y, $0274       ;When done initializing, fall into the main program loop.

; Index 0 is unreachable, not that you'd want to call it.
